Quick start: embed the publisher loader

  1. Create a slot under Ad slots. Wait until your publisher account is active and the slot is approved and active.
  2. Host publisher-snippet/src/loader.js on your site or CDN. The backend does not host this file.
  3. Put a slot element where the ad should appear. Set its ID to the slot's ad unit ID from the table above.
  4. Load Socket.IO before the loader. The loader requests and renders ads and sends impression, click, viewable impression, and report events.
<div data-ad-unit-id="YOUR_AD_UNIT_ID"></div>
<script src="https://cdn.socket.io/4.8.1/socket.io.min.js"></script>
<script src="https://your-cdn.example.com/loader.js"
        data-api-key="{{ publisher.api_key if publisher else 'YOUR_PUBLISHER_KEY' }}"
        data-server="{{ socket_base }}"></script>

Replace the sample loader URL and ad unit ID. The key is visible in browser source, so use it only on your registered site and do not reuse or share it elsewhere. Socket.IO requests check the origin against your registered site.

Advertisers need a creative with the exact width and height shown above, and that creative must be approved before it can serve.

Custom integration: Socket.IO

Use this if you are building your own integration instead of embedding the loader. Connect to the /delivery namespace, emit request_ad, then listen for serve_ad or no_fill.

<script src="https://cdn.socket.io/4.8.1/socket.io.min.js"></script>
<script>
const socket = io("{{ socket_base }}" + "/delivery", { transports: ["websocket"] });
const adUnitId = YOUR_AD_UNIT_ID;
const publisherKey = "{{ publisher.api_key if publisher else 'YOUR_PUBLISHER_KEY' }}";

socket.on("connect", () => socket.emit("request_ad", {
  ad_unit_id: adUnitId,
  api_key: publisherKey,
  context: { country: "IN", device_type: "mobile", page_keywords: ["travel"] }
}));
socket.on("serve_ad", ad => {
  const slot = document.querySelector("[data-ad-unit-id='" + adUnitId + "']");
  const link = document.createElement("a");
  link.href = ad.click_url;
  const image = new Image(ad.width, ad.height);
  image.src = ad.asset_url;
  image.alt = "Advertisement";
  link.appendChild(image);
  slot.replaceChildren(link);
  const eventData = () => ({ ad_unit_id: adUnitId, creative_id: ad.creative_id,
    visitor_id: ad.visitor_id, serve_token: ad.serve_token, event_id: crypto.randomUUID() });
  socket.emit("impression", eventData());
  link.addEventListener("click", () => socket.emit("click", eventData()));
});
socket.on("no_fill", info => console.log("No ad available", info.reason));
</script>

Include the returned visitor_id and serve_token on impressions and clicks. The token expires after five minutes; event IDs should be unique per event.

Custom integration: REST

Request an ad from your browser or server. Send the key in the X-Publisher-Key header. A successful match returns an ad; otherwise ad is null and reason explains the no-fill.

const API_BASE = "{{ api_base }}";
const publisherKey = "{{ publisher.api_key if publisher else 'YOUR_PUBLISHER_KEY' }}";
const adUnitId = YOUR_AD_UNIT_ID;

const response = await fetch(`${API_BASE}/delivery/ad/${adUnitId}?country=IN&device_type=mobile`, {
  headers: { "X-Publisher-Key": publisherKey }
});
const result = await response.json();
if (result.ad) {
  const ad = result.ad;
  const slot = document.querySelector("[data-ad-unit-id='" + adUnitId + "']");
  const link = document.createElement("a");
  link.href = ad.click_url;
  const image = new Image(ad.width, ad.height);
  image.src = ad.asset_url;
  image.alt = "Advertisement";
  link.appendChild(image);
  slot.replaceChildren(link);
  const track = (type) => fetch(`${API_BASE}/delivery/events/${type}`, {
    method: "POST", headers: { "Content-Type": "application/json", "X-Publisher-Key": publisherKey },
    body: JSON.stringify({ ad_unit_id: adUnitId, creative_id: ad.creative_id,
      visitor_id: ad.visitor_id, serve_token: ad.serve_token, event_id: crypto.randomUUID() })
  });
  track("impression");
  link.addEventListener("click", () => track("click"));
} else {
  console.log("No ad available", result.reason);
}

Record impressions and clicks with POST {{ api_base }}/delivery/events/impression or /delivery/events/click, using the same publisher-key header and a JSON body:

{
  "ad_unit_id": YOUR_AD_UNIT_ID,
  "creative_id": result.ad.creative_id,
  "event_id": crypto.randomUUID(),
  "visitor_id": result.ad.visitor_id,
  "serve_token": result.ad.serve_token
}

Do not put your publisher key in a URL. For public client-side placements, use HTTPS and restrict where you embed your integration.
